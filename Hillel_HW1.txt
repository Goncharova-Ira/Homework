'number' + 3 + 3
        відповідь: 'number33'
            до строки 'number' конкатенується число 3, вираз стає строкою, 
            і так само до строки конкатенується остання 3


null + 3 
        відповідь: 3
            null  стає числом 0 і до нього додається число 3

5 && "qwerty"
        відповідь: "qwerty"
            5!=0, тому набуває значення true,
            "qwerty" - не порожній рядок, тому він також -true
            результатом буде останнє  true

+'40' + +'2' + "hillel" 
        відповідь: '42hillel' 
            (+) перед строкою '40' та '2' перетворює їх на числа, отримуємо суму двох числел 42, 
            потім до них конкатенується строка 'hillel'

'10' - 5 === 6; 
        відповідь:false 
            строка '10' при відніманні перетворюється на число, отримуємо різницю двох чисел 10-5=5, 
            потім порівнюємо його з числом 6, вони не ріні, тому результат false

true + false = 1
            булієві значення перетворюються на числа, true-1? fslse -0, отримуємо суму 1+0=1

'4px' - 3 = NaN 
            строка '4px' не може бути перетворена у число, тому результат буде Not a Number

'4' - 3 = 1 
            строка перетвориться у число, отримаємо різницю двох чисел

'6' + 3 ** 0 = 61
            спочатку виконується піднесення до степеня: 3**0=1, 
            потім відбувається конкатенація строки '6' та 1

12 / '6' = 2 
            строка '6' перетворюється на число і відбуваєься ділення двох чисел

'10' + (5 === 6) = '10false' 
            спочатку відбувається порівняння в дужках з результатом false, 
            потім відбувається конкатенація строки '10' та 'false'

null == '' 
        відповідь: true 
            null  та порожня строка набувають значення false, тому вони рівні за значенням

3 ** (9 / 3) = 27 
            спочатку відбувається ділення в дужках 9/3=3, 
            потім відбувається піднесення до степеню

!!'false' == !!'true'
        результат: true
            обидва вирази є непорожньою строкою, тому вони будуть мати значення true, 
            потім перший знак ! обертає їх на false
            другий знак ! знов обертає їх на true, тож виходить true==true

0 || '0' && 1
        результат 1 
            спочатку виконується умова 'and'('0'&&1), строка '0' не порожня, тому вона true і 1 також true, тому результатом буде останнє true (1), 
            потім виконується умова 'or' (0||1), 0-це false, 1-true, результатом буде перше true, тобто (1)

(+null == false) < 1; 
        відповідь: false
            спершу виконується порівняння в дужках (+null == false), де +null перетворюється на число 0, false також перетворюється на 0, тобто рівність вірна - true, 
            потім відбувається порівняння 1<1- false

false && true || true 
        відповідь: true
            спочатку відбувається перевірка умови 'and', її результатом буде false, 
            потім перевіряється умова 'or' - false || true, результатом буде true

false && (false || true) 
        відповідь: false
            спочатку виконується дія в дужках, результатом буде true,
            перевіряється умова 'and' , її результатом буде false

(+null == false) < 1 ** 5; 
        відповідь: false
            в дужках +null перетвориться в 0 та false також перетвориться в 0 , тому рівність вірна true, тобто 1, 
            1**5 = 1
            1<1 - false
